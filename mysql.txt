-- 1. Tạo CSDL hỗ trợ tiếng Việt có dấu
CREATE DATABASE IF NOT EXISTS yhoc_corpus 
CHARACTER SET utf8mb4 
COLLATE utf8mb4_unicode_ci;

USE yhoc_corpus;

-- 2. Tạo bảng lưu trữ thông tin các bài báo y học được crawl về
CREATE TABLE IF NOT EXISTS articles (
    id INT AUTO_INCREMENT PRIMARY KEY,          
    title VARCHAR(500) NOT NULL,                
    authors TEXT,                               
    abstract TEXT,                              
    publication_year INT,                       
    source_url VARCHAR(512),                    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP 
);

-- 3. Tạo bảng lưu trữ nhật ký thu thập (logs)
CREATE TABLE IF NOT EXISTS crawl_logs (
    id INT AUTO_INCREMENT PRIMARY KEY,
    crawl_date DATE,
    target_url VARCHAR(512),
    total_urls INT,
    success_count INT,
    duplicate_count INT,
    error_count INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
