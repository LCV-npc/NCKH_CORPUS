CREATE DATABASE IF NOT EXISTS yhoc_corpus
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

USE yhoc_corpus;


-- ============================================================
-- 1. BẢNG BÀI BÁO
-- Lưu metadata, tóm tắt thu thập được và kết quả highlight NER.
-- ============================================================

CREATE TABLE IF NOT EXISTS articles (
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,

    title VARCHAR(500) NOT NULL,
    authors TEXT NULL,
    abstract LONGTEXT NULL,

    publication_year SMALLINT UNSIGNED NULL,
    source_url VARCHAR(512) NOT NULL,

    highlighted_html LONGTEXT NULL,

    created_at TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    PRIMARY KEY (id),

    UNIQUE KEY uq_articles_source_url (source_url),
    KEY idx_articles_publication_year (publication_year),
    KEY idx_articles_created_at (created_at),
    KEY idx_articles_title (title(191))
)
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_unicode_ci;

-- ============================================================
-- 2. BẢNG KHÁI NIỆM/THỰC THỂ ĐÃ GÁN NHÃN
-- Một bài báo có thể có nhiều thực thể.
-- ============================================================

CREATE TABLE IF NOT EXISTS extracted_concepts (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    article_id INT UNSIGNED NOT NULL,

    concept_name VARCHAR(500) NOT NULL,
    concept_type VARCHAR(100) NOT NULL,
    concept_code VARCHAR(100) NOT NULL DEFAULT '',
    concept_id VARCHAR(255) NOT NULL DEFAULT '',
    match_method VARCHAR(40) NOT NULL DEFAULT 'exact',
    confidence_score DECIMAL(6,5) NULL,
    dictionary_version VARCHAR(64) NULL,
    concept_start INT NULL,
    concept_end INT NULL,

    created_at TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (id),

    CONSTRAINT fk_extracted_concepts_article
        FOREIGN KEY (article_id)
        REFERENCES articles (id)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    KEY idx_concepts_article_id (article_id),
    KEY idx_concepts_type (concept_type),
    KEY idx_concepts_code (concept_code),
    KEY idx_concepts_concept_id (concept_id),
    KEY idx_concepts_match_method (match_method),
    KEY idx_concepts_name (concept_name(191)),
    KEY idx_concepts_top_statistics (
        concept_type,
        concept_name(191)
    )
)
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_unicode_ci;

-- ============================================================
-- Alias candidates discovered from the corpus.
-- Run this block manually before using the Dictionary Alias UI.
-- APPROVED rows are loaded by the NER engine at backend startup;
-- PENDING_REVIEW rows never affect highlighting.
-- ============================================================

CREATE TABLE IF NOT EXISTS dictionary_alias_candidates (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    surface_form VARCHAR(500) NOT NULL,
    concept_id VARCHAR(100) NOT NULL,
    canonical_term VARCHAR(500) NOT NULL,
    concept_code VARCHAR(100) NOT NULL,
    concept_type VARCHAR(100) NOT NULL,
    alias_type VARCHAR(40) NOT NULL DEFAULT 'grammar_variant',
    discovery_source VARCHAR(80) NOT NULL DEFAULT 'corpus',
    evidence_json LONGTEXT NULL,
    match_method VARCHAR(40) NOT NULL DEFAULT 'grammar_variant',
    confidence_score DECIMAL(6,5) NULL,
    uncertainty_score DECIMAL(6,5) NULL,
    occurrence_count INT UNSIGNED NOT NULL DEFAULT 0,
    document_frequency INT UNSIGNED NOT NULL DEFAULT 0,
    context_json LONGTEXT NULL,
    dictionary_version VARCHAR(64) NULL,
    status VARCHAR(24) NOT NULL DEFAULT 'PENDING_REVIEW',
    reviewed_by BIGINT UNSIGNED NULL,
    reviewed_at TIMESTAMP NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    PRIMARY KEY (id),
    UNIQUE KEY uq_dictionary_alias_surface_code (surface_form(191), concept_code),
    KEY idx_dictionary_alias_status (status, updated_at),
    KEY idx_dictionary_alias_concept (concept_id),
    KEY idx_dictionary_alias_method (match_method, status),
    KEY idx_dictionary_alias_reviewed_by (reviewed_by)
)
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


-- ============================================================
-- 3. BẢNG TIẾN TRÌNH THU THẬP
-- Dùng để dừng và tiếp tục tác vụ crawler.
-- Trạng thái hợp lệ trong backend:
-- running, paused, completed, error
-- ============================================================

CREATE TABLE IF NOT EXISTS crawl_progress (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

    target_url VARCHAR(512) NOT NULL,
    start_year SMALLINT UNSIGNED NOT NULL,
    end_year SMALLINT UNSIGNED NOT NULL,

    success_count INT UNSIGNED NOT NULL DEFAULT 0,
    duplicate_count INT UNSIGNED NOT NULL DEFAULT 0,
    skipped_count INT UNSIGNED NOT NULL DEFAULT 0,

    status VARCHAR(20) NOT NULL DEFAULT 'running',

    created_at TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    PRIMARY KEY (id),

    KEY idx_progress_resume (
        target_url,
        start_year,
        end_year,
        status
    ),

    KEY idx_progress_status (status),
    KEY idx_progress_created_at (created_at),

    CONSTRAINT chk_progress_year_range
        CHECK (start_year <= end_year),

    CONSTRAINT chk_progress_status
        CHECK (
            status IN (
                'running',
                'paused',
                'completed',
                'error'
            )
        )
)
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


-- ============================================================
-- 4. BẢNG NHẬT KÝ THU THẬP
-- Lưu thống kê cuối cùng của mỗi lần crawler chạy.
-- ============================================================

CREATE TABLE IF NOT EXISTS crawl_logs (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

    crawl_date DATE NOT NULL,
    target_url VARCHAR(512) NOT NULL,

    start_year SMALLINT UNSIGNED NULL,
    end_year SMALLINT UNSIGNED NULL,

    total_urls INT UNSIGNED NOT NULL DEFAULT 0,
    success_count INT UNSIGNED NOT NULL DEFAULT 0,
    duplicate_count INT UNSIGNED NOT NULL DEFAULT 0,
    error_count INT UNSIGNED NOT NULL DEFAULT 0,

    status VARCHAR(20) NOT NULL DEFAULT 'completed',
    stopped_at TIMESTAMP NULL DEFAULT NULL,

    created_at TIMESTAMP NOT NULL
        DEFAULT CURRENT_TIMESTAMP,

    PRIMARY KEY (id),

    KEY idx_crawl_logs_created_at (created_at),
    KEY idx_crawl_logs_date (crawl_date),
    KEY idx_crawl_logs_status (status),
    KEY idx_crawl_logs_target_year (
        target_url,
        start_year,
        end_year
    ),

    CONSTRAINT chk_crawl_logs_year_range
        CHECK (
            start_year IS NULL
            OR end_year IS NULL
            OR start_year <= end_year
        ),

    CONSTRAINT chk_crawl_logs_status
        CHECK (
            status IN (
                'running',
                'paused',
                'completed',
                'error'
            )
        )
)
ENGINE = InnoDB
DEFAULT CHARACTER SET = utf8mb4
COLLATE = utf8mb4_unicode_ci;

-- ============================================================
-- 6. CORPUS REVIEW: AUTHENTICATION, AI LABELING, EXPERT AUDIT
-- These tables extend the existing corpus schema. They do not alter or
-- duplicate articles/extracted_concepts. The backend runs the same
-- CREATE TABLE IF NOT EXISTS migration at startup.
-- ============================================================

CREATE TABLE IF NOT EXISTS users (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    full_name VARCHAR(200) NOT NULL,
    email VARCHAR(254) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(16) NOT NULL DEFAULT 'EXPERT',
    is_active TINYINT(1) NOT NULL DEFAULT 1,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_users_email (email),
    KEY idx_users_role (role),
    CONSTRAINT chk_users_role CHECK (role IN ('ADMIN', 'EXPERT', 'REVIEWER'))
)
ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS user_sessions (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    user_id BIGINT UNSIGNED NOT NULL,
    token_hash CHAR(64) NOT NULL,
    expires_at DATETIME NOT NULL,
    last_seen_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    revoked_at DATETIME NULL DEFAULT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_user_sessions_token_hash (token_hash),
    KEY idx_user_sessions_user (user_id),
    KEY idx_user_sessions_expiry (expires_at),
    CONSTRAINT fk_user_sessions_user FOREIGN KEY (user_id)
        REFERENCES users(id) ON UPDATE CASCADE ON DELETE CASCADE
)
ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS ai_document_labels (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    article_id INT UNSIGNED NOT NULL,
    model_name VARCHAR(120) NOT NULL,
    label_payload LONGTEXT NOT NULL,
    primary_icd10_code VARCHAR(100) NULL,
    primary_icd10_label VARCHAR(500) NULL,
    confidence DECIMAL(5,4) NULL,
    generated_by BIGINT UNSIGNED NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    KEY idx_ai_labels_article (article_id, id),
    KEY idx_ai_labels_primary_code (primary_icd10_code),
    CONSTRAINT fk_ai_labels_article FOREIGN KEY (article_id)
        REFERENCES articles(id) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_ai_labels_user FOREIGN KEY (generated_by)
        REFERENCES users(id) ON UPDATE CASCADE ON DELETE SET NULL
)
ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS labeling_sessions (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    name VARCHAR(255) NOT NULL,
    description TEXT NULL,
    reviewer_id BIGINT UNSIGNED NOT NULL,
    status VARCHAR(24) NOT NULL DEFAULT 'OPEN',
    created_by BIGINT UNSIGNED NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    KEY idx_labeling_sessions_status (status),
    KEY idx_labeling_sessions_reviewer (reviewer_id),
    CONSTRAINT chk_labeling_sessions_status CHECK (status IN ('OPEN', 'CLOSED', 'CANCELLED')),
    CONSTRAINT fk_labeling_sessions_reviewer FOREIGN KEY (reviewer_id)
        REFERENCES users(id) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_labeling_sessions_creator FOREIGN KEY (created_by)
        REFERENCES users(id) ON UPDATE CASCADE ON DELETE RESTRICT
)
ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS labeling_session_assignments (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    session_id BIGINT UNSIGNED NOT NULL,
    article_id INT UNSIGNED NOT NULL,
    expert_id BIGINT UNSIGNED NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_session_article_expert (session_id, article_id, expert_id),
    KEY idx_assignment_expert (expert_id),
    KEY idx_assignment_article (article_id, session_id),
    CONSTRAINT fk_assignment_session FOREIGN KEY (session_id)
        REFERENCES labeling_sessions(id) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_assignment_article FOREIGN KEY (article_id)
        REFERENCES articles(id) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_assignment_expert FOREIGN KEY (expert_id)
        REFERENCES users(id) ON UPDATE CASCADE ON DELETE RESTRICT
)
ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS expert_reviews (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    document_id INT UNSIGNED NOT NULL,
    session_id BIGINT UNSIGNED NULL,
    expert_id BIGINT UNSIGNED NOT NULL,
    review_status VARCHAR(32) NOT NULL,
    -- 'icd10' = nhãn từ NER/từ điển (extracted_concepts)
    -- 'ai'    = nhãn từ AI Gemini (ai_document_labels)
    label_source VARCHAR(16) NOT NULL DEFAULT 'icd10'
        COMMENT 'icd10 = NER dictionary, ai = AI label',
    original_labels_json LONGTEXT NOT NULL,
    annotations_json LONGTEXT NULL,
    suggested_icd10_code VARCHAR(100) NULL,
    suggested_icd10_label VARCHAR(500) NULL,
    comment TEXT NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    KEY idx_reviews_document (document_id, id),
    KEY idx_reviews_session (session_id, document_id, expert_id),
    KEY idx_reviews_expert (expert_id, id),
    KEY idx_reviews_status (review_status),
    CONSTRAINT fk_reviews_article FOREIGN KEY (document_id)
        REFERENCES articles(id) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_reviews_session FOREIGN KEY (session_id)
        REFERENCES labeling_sessions(id) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_reviews_expert FOREIGN KEY (expert_id)
        REFERENCES users(id) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT chk_reviews_status CHECK (
        review_status IN ('CORRECT', 'INCORRECT', 'NEEDS_REVISION')
    ),
    CONSTRAINT chk_reviews_label_source CHECK (
        label_source IN ('icd10', 'ai')
    )
)
ENGINE = InnoDB DEFAULT CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS reviewer_adjudications (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    document_id INT UNSIGNED NOT NULL,
    session_id BIGINT UNSIGNED NULL,
    reviewer_id BIGINT UNSIGNED NOT NULL,
    expert_review_a_id BIGINT UNSIGNED NULL,
    expert_review_b_id BIGINT UNSIGNED NULL,
    decision_payload LONGTEXT NOT NULL,

    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    KEY idx_adjudications_document (document_id, id),

    CONSTRAINT fk_adjudications_article FOREIGN KEY (document_id)
        REFERENCES articles(id) ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT fk_adjudications_session FOREIGN KEY (session_id)
        REFERENCES labeling_sessions(id) ON UPDATE CASCADE ON DELETE RESTRICT,

    CONSTRAINT fk_adjudications_review_a FOREIGN KEY (expert_review_a_id)

    CONSTRAINT fk_adjudications_review_b FOREIGN KEY (expert_review_b_id)

)

-- ============================================================
-- 7. LANGUAGE ADMISSION AUDIT
-- Every OJS crawler decision is stored independently of articles so rejected
-- English/mixed/no-text candidates remain traceable without entering corpus.
-- The backend creates this table automatically on startup as well.
-- ============================================================

CREATE TABLE IF NOT EXISTS corpus_language_audit (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    article_id INT UNSIGNED NULL,
    source_url VARCHAR(512) NOT NULL,
    source_domain VARCHAR(255) NOT NULL DEFAULT '',
    pdf_url VARCHAR(1024) NULL,
    article_title VARCHAR(500) NULL,
    decision_status VARCHAR(40) NOT NULL,
    rejection_reason VARCHAR(160) NOT NULL DEFAULT '',
    metadata_language VARCHAR(20) NOT NULL DEFAULT 'unknown',
    metadata_confidence DECIMAL(6,4) NULL,
    pdf_language VARCHAR(20) NULL,
    pdf_confidence DECIMAL(6,4) NULL,
    vietnamese_ratio DECIMAL(6,4) NULL,
    english_ratio DECIMAL(6,4) NULL,
    assessed_characters INT UNSIGNED NULL,
    file_path VARCHAR(1024) NULL,
    file_hash_sha256 CHAR(64) NULL,
    evidence_json LONGTEXT NOT NULL,
    first_seen_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    last_seen_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_corpus_language_audit_source (source_url),
    KEY idx_corpus_language_audit_status (decision_status),
    KEY idx_corpus_language_audit_article (article_id),
    KEY idx_corpus_language_audit_hash (file_hash_sha256)
) ENGINE=InnoDB DEFAULT CHARACTER SET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
